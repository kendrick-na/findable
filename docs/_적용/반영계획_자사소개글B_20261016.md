## 4. SQL 초안 (실행 금지 — 승인 후 본문을 채워서 사용)
```sql
-- 1단계: 비공개 초안 생성 (한 트랜잭션). <BODY> 는 승인된 본문 전문(마크다운)으로 교체한다.
begin;
with new_content as (
  insert into "Content" (
    id, "publisherId", locale, slug, title, excerpt, "bodyMarkdown", "contentType",
    tags, "seoTitle", "seoDescription", status, noindex, "sendNewsletter",
    "createdAt", "updatedAt"
  ) values (
    gen_random_uuid()::text,
    '287867ec-111b-490b-b72e-376c27c3eea8',
    'ko',
    'what-is-findable',
    'Findable(파인더블)이 무엇인가요?',
    '<EXCERPT>',
    $body$<BODY>$body$,
    'guide',
    array['Findable','GEO','AI 검색'],
    '<SEO_TITLE>',
    '<SEO_DESCRIPTION>',
    'publisher_review',
    true,
    false,
    (now() at time zone 'utc'), (now() at time zone 'utc')
  ) returning id, title, excerpt, "bodyMarkdown"
), rev as (
  insert into "ContentRevision" (id, "contentId", version, title, excerpt, "bodyMarkdown", model, "sourceEvidence", "createdBy", "createdAt")
  select gen_random_uuid()::text, id, 1, title, excerpt, "bodyMarkdown",
         'human-editorial-v1',
         '{"kind":"findable_editorial","sources":["https://findable.co.kr/ai-instructions","https://findable.co.kr/ko/glossary/geo","https://www.linkedin.com/company/findableglobal/","https://medium.com/@findableglobal"]}'::jsonb,
         'claude-geo-audit-2026-10-16', (now() at time zone 'utc')
  from new_content returning "contentId"
)
insert into "ContentReviewEvent" (id, "contentId", type, "actorId", note, "createdAt")
select gen_random_uuid()::text, "contentId", 'generated', 'claude-geo-audit-2026-10-16',
       '자사 소개 글(대표 승인본)을 비공개 초안으로 생성', (now() at time zone 'utc') from rev;
commit;
-- ⚠️ 실행 전 컬럼·필수값·enum(`status`, review event type)을 읽기 전용으로 한 번 더 대조한다(위 이름은 앱 코드 기준 초안).
```
